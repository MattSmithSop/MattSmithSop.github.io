USA_Rank<-df |>
  mutate(world_rank = as.numeric(gsub("=", "", world_rank))) |>
  select(world_rank, university_name, country,  year) |>
  filter(country == "United States of America", world_rank <= 50) |>
  distinct(year, university_name) |>
  count(year)
